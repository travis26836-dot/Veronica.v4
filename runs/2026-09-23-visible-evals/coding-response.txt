def unique_in_order(items):
    seen = set()
    result = []
    for item in items:
        if item not in seen:
            seen.add(item)
            result.append(item)
    return result

# Tests
test1 = unique_in_order([1, 2, 2, 3, 1])
assert test1 == [1, 2, 3, 1], "Test 1 failed"

test2 = unique_in_order(['a', 'b', 'a', 'c', 'b'])
assert test2 == ['a', 'b', 'c'], "Test 2 failed"

test3 = unique_in_order([])
assert test3 == [], "Test 3 failed"
